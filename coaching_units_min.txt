def load_root(root, config):
 seen = set()
 matches = []
 for folder in find_analyses(root):
  manifest = json.loads((folder / MANIFEST_NAME).read_text(encoding='utf-8'))
  identity = str(manifest['analysis']['video_identity'])
  if identity in seen:
   continue
  seen.add(identity)
  matches.append(_load_match(folder, root, identity, manifest, config))
 return matches
def _load_match(folder, root, identity, manifest, config):
 _, scoring = _score(folder, config)
 name = str(folder.relative_to(root))
 thresholds = config.coach.death_factor_thresholds
 deaths = []
 for unit in scoring.units:
  coach_input = scoring.coach_inputs[unit.candidate_id]
  view = build_coach_llm_view(coach_input, unit, thresholds)
  deaths.append(observation(name, unit, coach_input, view))
 scenarios = json.loads((folder / 'scenarios.json').read_text(encoding='utf-8'))
 rows = scenarios if isinstance(scenarios, list) else scenarios.get('scenarios', [])
 return MatchData(name=name, identity=identity, deaths=deaths, engagements=[r for r in rows if r['scenario_type'] == 'engagement'], low_ink=[(float(e['start_time']), float(e.get('end_time') or e['start_time'])) for e in manifest.get('game_events') or () if e.get('event_type') == 'low_ink'], death_intervals=[(float(r['start_time']), float(r['end_time'])) for r in rows if r['scenario_type'] == 'death_episode'])
def observation(name, unit, coach_input, view):
 payload = view.model_dump(mode='json')
 timeline = coach_input.primary_context.timeline
 gap = timeline.time_since_previous_death if timeline else None
 t = float(unit.video_time)
 map_block = (payload.get('map') or {}).get('map_check_before_death')
 basis = list(payload.get('claim_basis') or ())
 return Obs(match=name, cid=unit.candidate_id, t=t, score=float(unit.importance_score), factors=frozenset((f.factor_id for f in unit.factors if f.contribution > 0)), observed=frozenset((f.factor_id for f in unit.factors if f.active)), prev_death_t=None if gap is None else round(t - float(gap), 3), selected=bool(unit.selected_for_llm), prompt=serialize_llm_view_user_prompt(view), body=json.dumps(payload, ensure_ascii=False, sort_keys=True, separators=(',', ':')), coachable=frozenset((e['claim_id'] for e in basis if e.get('coachable'))), templates=frozenset((r for e in basis for r in e.get('rec_template_ids') or ())), statements=tuple((f.statement_player for f in unit.factors if f.active and f.statement_player)), special=(payload.get('special') or {}).get('state_at_anchor'), map_status=map_block.get('status') if isinstance(map_block, dict) else None)
def pool(match, policy, gap, modifiers):
 return candidates(match, policy, gap, modifiers)
def by_factor(scoring, modifiers):
 groups = {}
 for death in scoring:
  groups.setdefault(primary(death, modifiers), []).append(death)
 return [Candidate(members) for members in groups.values()]
def multi_scoring_episodes(match, gap):
 return [g for g in episodes(match, gap) if sum((d.score > 0 for d in g)) >= 2]
def unit_prompt(unit):
 lead = unit.lead
 others = sorted((d for d in unit.members if d is not lead), key=lambda d: d.t)
 if not others:
  return lead.prompt
 return lead.prompt + OBSERVATIONS_HEADER + '\n'.join((d.body for d in others))
def zero_score_members(units):
 return [d for u in units if u.score > 0 or len(u.members) > 1 for d in u.members if d.score <= 0]
def references(match):
 by_t = {round(d.t, 3): d for d in match.deaths}
 pairs = []
 for death in match.deaths:
  if death.score <= 0 or death.prev_death_t is None:
   continue
  prev = by_t.get(death.prev_death_t)
  if prev is not None and prev.score <= 0:
   pairs.append((death, prev))
 return pairs
def _episode_index(match, gap):
 return {d.cid: i for i, group in enumerate(episodes(match, gap)) for d in group}
def _signature(unit, modifiers):
 return frozenset((primary(d, modifiers) for d in unit.members))
def redundant_units(match, picks, gap, modifiers):
 index = _episode_index(match, gap)
 flagged = []
 for i, unit in enumerate(picks):
  templates = frozenset().union(*(d.templates for d in unit.members))
  for earlier in picks[:i]:
   same_episode = index[earlier.lead.cid] == index[unit.lead.cid]
   shared = _signature(unit, modifiers) & _signature(earlier, modifiers)
   earlier_templates = frozenset().union(*(d.templates for d in earlier.members))
   if same_episode and shared or (templates and templates == earlier_templates):
    flagged.append(unit)
    break
 return flagged
def _unit_context(unit):
 claims = frozenset().union(*(d.coachable for d in unit.members))
 return (len(claims), len(claims - unit.lead.coachable), len(unit.members) - 1)
def policy_metrics(matches, policy, gap, top, modifiers):
 pools = {m.name: pool(m, policy, gap, modifiers) for m in matches}
 picks = {name: select(units, top) for name, units in pools.items()}
 positive = {d.key for m in matches for d in m.deaths if d.score > 0}
 in_pool = {d.key for units in pools.values() for u in units for d in u.members}
 selected = [u for units in picks.values() for u in units]
 in_selected = {d.key for u in selected for d in u.members}
 context = [_unit_context(u) for u in selected]
 prompts = [len(unit_prompt(u)) for u in selected]
 by_name = {m.name: m for m in matches}
 return {'policy': policy, 'pool_units': sum((len(v) for v in pools.values())), 'selected_units': len(selected), 'matches_covered': sum((1 for v in picks.values() if v)), 'positive_total': len(positive), 'positive_as_observations': len(positive & in_pool), 'positive_in_selected': len(positive & in_selected), 'zero_promoted': len(zero_score_members((u for units in pools.values() for u in units))), 'redundant_units': sum((len(redundant_units(by_name[n], v, gap, modifiers)) for n, v in picks.items())), 'distinct_opportunities': sum((len({primary(d, modifiers) for u in v for d in u.members} - {frozenset()}) for v in picks.values())), 'context_claims_per_unit': _mean([c[0] for c in context]), 'claims_lost_lead_only': sum((c[1] for c in context)), 'anchors_lost_lead_only': sum((c[2] for c in context)), 'prompt_chars_total': sum(prompts), 'prompt_chars_per_unit': _mean(prompts), 'max_calls_per_match': max((len(v) for v in picks.values()), default=0)}
def _mean(values):
 return round(statistics.fmean(values), 2) if values else 0.0
def gap_sweep(matches, top, modifiers):
 rows = []
 for gap in GAP_SWEEP:
  groups = [g for m in matches for g in episodes(m, gap)]
  multi = [g for g in groups if len(g) >= 2]
  row = {'gap': gap, 'episodes_multi_death': len(multi), 'episodes_multi_scoring': sum((1 for g in multi if sum((d.score > 0 for d in g)) >= 2)), 'longest_span': max((g[-1].t - g[0].t for g in multi), default=0.0)}
  for policy in POLICIES:
   row[policy] = sum((len(select(pool(m, policy, gap, modifiers), top)) for m in matches))
  rows.append(row)
 return rows
def _selected_times(matches, policy, gap, top, modifiers):
 out = {}
 for m in matches:
  picks = select(pool(m, policy, gap, modifiers), top)
  out[m.identity] = sorted((d.t for u in picks for d in u.members))
 return out
def jaccard_by_time(left, right, tolerance):
 shared = sorted(set(left) & set(right))
 union = inter = 0
 for identity in shared:
  a, b = (list(left[identity]), list(right[identity]))
  hits = 0
  for t in a:
   near = [x for x in b if abs(x - t) <= tolerance]
   if near:
    b.remove(min(near, key=lambda x: abs(x - t)))
    hits += 1
  inter += hits
  union += len(left[identity]) + len(right[identity]) - hits
 return round(inter / union, 3) if union else None
def stability(roots, pairs, top, gap, modifiers):
 rows = []
 for a, b in pairs:
  if a not in roots or b not in roots:
   continue
  row = {'pair': f'{a} vs {b}'}
  shared = {m.identity for m in roots[a]} & {m.identity for m in roots[b]}
  row['shared_matches'] = len(shared)
  for policy in POLICIES:
   left = _selected_times(roots[a], policy, gap, top, modifiers)
   right = _selected_times(roots[b], policy, gap, top, modifiers)
   row[policy] = jaccard_by_time(left, right, STABILITY_TOLERANCE_SECONDS)
  rows.append(row)
 return rows
def _slug(name):
 return re.sub('[^A-Za-z0-9]+', '_', name).strip('_')[:48]
def deferred_arms(scoring, modifiers):
 return {'individual': [Candidate([d]) for d in scoring], 'episode': [Candidate(list(scoring))], 'episode_by_factor': by_factor(scoring, modifiers)}
def deferred_inputs(matches, gap, modifiers, system_prompt):
 files = {'system_prompt.txt': system_prompt}
 episodes_meta = []
 for match in matches:
  for i, group in enumerate(multi_scoring_episodes(match, gap), start=1):
   scoring = [d for d in group if d.score > 0]
   slug = f'{_slug(match.name)}__ep{i}'
   arms_meta = {}
   for arm, units in deferred_arms(scoring, modifiers).items():
    arms_meta[arm] = []
    for j, unit in enumerate(sorted(units, key=lambda u: u.lead.t), start=1):
     path = f'{slug}/{arm}/call_{j}.user_prompt.txt'
     files[path] = unit_prompt(unit)
     arms_meta[arm].append({'file': path, 'chars': len(files[path]), 'lead': unit.lead.cid, 'members': [d.cid for d in sorted(unit.members, key=lambda d: d.t)]})
   episodes_meta.append({'match': match.name, 'episode': slug, 'arms': arms_meta})
 manifest = {'settings': DEFERRED_SETTINGS, 'system_prompt_chars': len(system_prompt), 'episodes': episodes_meta}
 files['manifest.json'] = json.dumps(manifest, indent=1, sort_keys=True) + '\n'
 return files
def write_files(out_dir, files):
 if out_dir.is_dir():
  for path in sorted(out_dir.rglob('*'), reverse=True):
   path.unlink() if path.is_file() else path.rmdir()
 for rel, text in sorted(files.items()):
  path = out_dir / rel
  path.parent.mkdir(parents=True, exist_ok=True)
  path.write_text(text, encoding='utf-8')
def rating_units(matches, gap, top, modifiers):
 units = {}
 for policy in POLICIES:
  for match in matches:
   for unit in select(pool(match, policy, gap, modifiers), top):
    key = tuple(sorted((d.key for d in unit.members)))
    entry = units.setdefault(key, {'unit': unit, 'policies': []})
    entry['policies'].append(policy)
 return units
def _evidence_text(unit):
 return ' || '.join((f'{d.t:.1f}s: ' + (' '.join(d.statements) or '(no factor statement)') for d in sorted(unit.members, key=lambda d: d.t)))
def rating_sheet(matches, gap, top, modifiers, seed=RATING_SEED):
 units = rating_units(matches, gap, top, modifiers)
 keys = sorted(units)
 random.Random(seed).shuffle(keys)
 sheet, key = (io.StringIO(), io.StringIO())
 sheet_w = csv.writer(sheet, lineterminator='\n')
 key_w = csv.writer(key, lineterminator='\n')
 sheet_w.writerow(('unit_id', 'match', 'anchors', 'evidence', *RATING_COLUMNS))
 key_w.writerow(('unit_id', 'match', 'members', 'lead', 'policies'))
 for n, members in enumerate(keys, start=1):
  unit = units[members]['unit']
  unit_id = f'unit_{n:03d}'
  anchors = '; '.join((f'{d.t:.1f}s' for d in sorted(unit.members, key=lambda d: d.t)))
  sheet_w.writerow((unit_id, unit.lead.match, anchors, _evidence_text(unit), *[''] * 6))
  key_w.writerow((unit_id, unit.lead.match, ';'.join((d.cid for d in unit.members)), unit.lead.cid, ';'.join(units[members]['policies'])))
 return (sheet.getvalue(), key.getvalue())

def _engagement_stats(matches):
 rows = [e for m in matches for e in m.engagements]
 durations = [float(e['end_time']) - float(e['start_time']) for e in rows]
 splats = [int((e.get('context') or {}).get('splat_count') or 0) for e in rows]
 linked = sum((1 for e in rows if (e.get('context') or {}).get('following_death_id')))
 return {'count': len(rows), 'median_duration': statistics.median(durations) if durations else 0.0, 'single_splat': sum((1 for s in splats if s <= 1)), 'linked_to_death': linked}
def _overlaps(a, b):
 return a[0] <= b[1] and b[0] <= a[1]
def _low_ink_stats(matches):
 total = free = 0
 for m in matches:
  for interval in m.low_ink:
   total += 1
   if not any((_overlaps(interval, d) for d in m.death_intervals)):
    free += 1
 return {'count': total, 'without_death': free}

def _contradictions(group):
 scoring = [d for d in group if d.score > 0]
 notes = []
 for label, attr in (('special', 'special'), ('map', 'map_status')):
  states = {getattr(d, attr) for d in scoring}
  if len(states) > 1:
   notes.append(f'{label} differs: ' + ', '.join((f'{d.t:.1f}s={getattr(d, attr)}' for d in scoring)))
 return notes
def experiment_b(matches, gap):
 lines = ['## Experiment B: grouping semantics', '', f'Episodes with two or more positive-scoring deaths (link gap {gap:g}s):', '', '| match | members (time factors = score) | distinct primary sets | span | member disagreement |', '| --- | --- | ---: | ---: | --- |']

  for group in multi_scoring_episodes(m, gap):
   scoring = [d for d in group if d.score > 0]
   distinct = len({d.factors - {'death_final_30s', 'death_first_30s'} for d in scoring})
   lines.append(f"| {m.name[:36]} | {'; '.join((_member(d) for d in group))} | {distinct} | {group[-1].t - group[0].t:.1f}s | {'; '.join(_contradictions(group)) or 'none'} |")

def _label(match, group):
 scoring = [d for d in group if d.score > 0]
 return f"{match.name[:36]} ({', '.join((_member(d) for d in scoring))})"
def _primary_sets(group):
 clock = {'death_final_30s', 'death_first_30s'}
 return {d.factors - clock for d in group if d.score > 0}
def _episode_examples(matches, gap):
 found = [(m, g) for m in matches for g in multi_scoring_episodes(m, gap)]
 longest = max(found, key=lambda mg: mg[1][-1].t - mg[1][0].t, default=None)
 mixed = [_label(m, g) for m, g in found if len(_primary_sets(g)) > 1]
 same = [_label(m, g) for m, g in found if len(_primary_sets(g)) == 1]
 span = f'{longest[0].name[:36]} links {len(longest[1])} deaths across {longest[1][-1].t - longest[1][0].t:.1f}s' if longest else 'no multi-death episode exists'
 return {'span': span, 'mixed': '; '.join(mixed) or 'none in this corpus', 'same': '; '.join(same) or 'none in this corpus'}

@dataclass
class Candidate:
 members: int
 @property
 def lead(self):
  return min(self.members, key=lambda d: (-d.score, d.t, d.cid))
 @property
 def score(self):
  return self.lead.score
 def label(self, modifiers):
  return ' + '.join((f'{d.t:.0f}s {_short(d.factors, modifiers)}' for d in sorted(self.members, key=lambda d: d.t)))
def _short(factors, modifiers):
 primary = sorted((SHORT.get(f, f) for f in factors if f not in modifiers))
 extra = sorted((SHORT.get(f, f) for f in factors if f in modifiers))
 return '/'.join(primary) + (f" (+{'/'.join(extra)})" if extra else '') or '–'
def primary(death, modifiers):
 return frozenset((f for f in death.factors if f not in modifiers))
def episodes(match, gap):
 groups = []
 for death in sorted(match.deaths, key=lambda d: d.t):
  if groups and death.t - groups[-1][-1].t <= gap:
   groups[-1].append(death)
  else:
   groups.append([death])
 return groups
def candidates(match, policy, gap, modifiers):
 if policy == 'deaths':
  return [Candidate([d]) for d in match.deaths]
 out = []
 for group in episodes(match, gap):
  scoring = [d for d in group if d.score > 0]
  if not scoring:
   out.extend((Candidate([d]) for d in group))
  elif policy == 'episode':
   out.append(Candidate(scoring))
  else:
   by_factor = {}
   for death in scoring:
    by_factor.setdefault(primary(death, modifiers), []).append(death)
   out.extend((Candidate(members) for members in by_factor.values()))
 return out

### tools/README.md
# tools/
Developer tooling that lives **outside** the installable `splatoon3_ai_coach`
package. Nothing in `src/` may import from `tools/` except the CLI commands that
launch the web apps below.
Run scripts from the repository root with the project virtualenv:
```bash
.venv/bin/python tools/<folder>/<script>.py --help
```
Most scripts read local analysis output under `analysis/` (gitignored), so they
need you to have run `s3-coach analyze` on your own recordings first.
## Layout
| Folder | What it holds | Maintained? |
|--------|---------------|-------------|
| `calibration/` | Interactive/offline helpers for drawing and checking ROIs, stage masks, and colors | Yes |
| `diagnostics/` | Validators and diagnostics run against analysis artifacts | Yes |
| `reports/` | Death-importance ranking reports over a library of analyzed matches | Yes |
| `studies/` | One-off research scripts that informed a design decision | **No** (see below) |
| `vision_manifest_viewer/` | Read-only HTML viewer for `vision_manifest.json` (`s3-coach vision-view`) | Yes |
| `vmv_site/` | Local analysis platform API (`s3-coach vmv-site`) | Yes |
| `public_site/` | Public coaching submission site (`s3-coach public-site`) | Yes |
The three web packages stay at the top level because the CLI imports them by
putting `tools/` on `sys.path`.
### calibration/
| Script | Purpose |
|--------|---------|
| `roi_calibrate.py` | Interactive ROI calibration for normalized `[x1,y1,x2,y2]` boxes (shared helpers used by the other calibration scripts) |
| `roi_visualize.py` | Draw normalized ROI boxes on an image or video frame |
| `score_roi_calibrate.py` | Calibrate Splat Zones score digit ROIs |
| `stage_mask_calibrate.py` | Interactive playable-stage polygon calibration |
| `stage_mask_ab_compare.py` | A/B paint %: stage polygon mask vs ROI-union fallback |
| `map_ink_roi_preview.py` | Preview stage-map ROI coverage on a frame |
| `map_ink_color_audit.py` | Offline map ink color audit |
| `team_color_probe.py` | Print the team color probe for a `vision_manifest.json` |
| `review_special_icon_harvest.py` | Crop Review event-lane icon candidates with source and player-slot metadata |
### diagnostics/
| Script | Purpose |
|--------|---------|
| `validate_review_timeline.py` | Deterministic GO/NO-GO quality gate for a Review timeline artifact |
| `map_ink_geometry_validate.py` | Multi-frame map ink geometry validation (needs local paint-map references) |
| `special_gauge_diagnose.py` | Special gauge diagnostics on survey representative frames |
| `trace_fixture.py` | Writes a trimmed golden-trace fixture (`tests/fixtures/traces/`) from a real analysis directory |
| `claim_corpus.py` | Freezes saved LLM outputs into the claim-validity corpus (`tests/fixtures/claim_corpus/`) and reports per-rule metrics under current validation |
### reports/
`importance_report.py` provides shared loaders used by the `ranking_*.py`
scripts, so keep these together.
| Script | Purpose |
|--------|---------|
| `importance_report.py` | Re-score every analyzed match and compare with its last ranking |
| `ranking_competition.py` | Competition among death candidates in the importance ranking |
| `ranking_cases.py` | Per-death case analysis of the importance factors |
| `ranking_episodes.py` | Deaths versus coaching episodes as the selection unit |
| `coaching_units.py` | Research comparison of coaching units (deaths, episodes, episode-by-factor; engagement and non-death feasibility); writes `coaching_units.md`, a blind rating sheet with a separate key, and the deferred Experiment D inputs (never calls a model). Sections live in `coaching_units_sections.py` |
### studies/
Research scripts kept so past findings stay reproducible. They are **not
maintained**: they may hardcode analysis folder names, expect specific local
recordings, or describe a stage of work that has since shipped. Do not import
them from new code. If a study's logic becomes part of the product, move it into
`src/` with tests instead of extending the script.
Studies cover Splat Zones / Clam Blitz score reading and trajectories, zone
control, special activation, the Review-feature survey (Phase 0), and score
fusion validation.
## Adding a tool
- Put it in the folder that matches its purpose; prefer `diagnostics/` for
  anything that checks an artifact and can fail.
- Resolve the repository root with
  `from splatoon3_ai_coach.config.paths import PROJECT_ROOT` rather than
  counting `Path(__file__).parents`.
- Use `argparse` (or Typer) so `--help` works without local data.
- If tests import it, the folder must be listed in `pythonpath` under
  `[tool.pytest.ini_options]` in `pyproject.toml` (all four folders already are).
